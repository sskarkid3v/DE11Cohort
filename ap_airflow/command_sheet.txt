# 1) create an .env with your uid so file perms are correct
printf "AIRFLOW_UID=%s\nAIRFLOW_PROJ_DIR=.\nAIRFLOW_IMAGE_NAME=apache/airflow:3.0.6\n" "$(id -u)" > .env

# 2) create folders for mounts
mkdir -p dags logs plugins config
sudo chown -R $(id -u):0 dags logs plugins config
sudo chmod -R g+w dags logs plugins config

# 3) (optional) generate and export a Fernet key
python3 - <<'PY'
import base64, os
print(base64.urlsafe_b64encode(os.urandom(32)).decode())
PY
export AIRFLOW__CORE__FERNET_KEY="<paste-output>"

# 4) pull images and start the init container (sets up DB, users, perms)
docker compose up airflow-init

# 5) once airflow-init exits with 0, start all services
docker compose up -d

# 6) verify services
docker compose ps
docker compose logs -f airflow-apiserver   # main UI/API
docker compose logs -f airflow-scheduler   # scheduler
docker compose logs -f airflow-worker      # celery worker
docker compose logs -f airflow-dag-processor
